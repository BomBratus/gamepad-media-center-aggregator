cmake_minimum_required(VERSION 3.18)
project(gmca_imdb_tests C CXX)
find_package(Threads REQUIRED)
include(../../cmake/imdb_dependencies.cmake)
add_executable(test_imdb_index ../test_imdb_index.cpp ../../app/src/api/stremio/imdb_index.cpp)
target_compile_features(test_imdb_index PRIVATE cxx_std_17)
target_compile_options(test_imdb_index PRIVATE -Wall -Wextra -UNDEBUG)
target_compile_definitions(test_imdb_index PRIVATE GMCA_INDEX_NOLOCK)
target_include_directories(test_imdb_index PRIVATE ../support ../../app/include
    "${GMCA_JSON_INCLUDE}")
target_link_libraries(test_imdb_index PRIVATE gmca_sqlite gmca_index_zlib Threads::Threads ${CMAKE_DL_LIBS} m)
add_executable(test_archive_playback_gate ../test_archive_playback_gate.cpp)
target_compile_features(test_archive_playback_gate PRIVATE cxx_std_17)
target_compile_options(test_archive_playback_gate PRIVATE -Wall -Wextra -UNDEBUG)
target_include_directories(test_archive_playback_gate PRIVATE ../../app/include)
target_link_libraries(test_archive_playback_gate PRIVATE Threads::Threads)
enable_testing()
add_executable(test_archive_query_queue ../test_archive_query_queue.cpp)
target_compile_features(test_archive_query_queue PRIVATE cxx_std_17)
target_compile_options(test_archive_query_queue PRIVATE -Wall -Wextra -UNDEBUG)
target_include_directories(test_archive_query_queue PRIVATE ../../app/include)
target_link_libraries(test_archive_query_queue PRIVATE Threads::Threads)
add_test(NAME archive_query_queue COMMAND test_archive_query_queue)
add_test(NAME imdb_index COMMAND test_imdb_index)
add_test(NAME archive_playback_gate COMMAND test_archive_playback_gate)
